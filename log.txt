        Quét system_activity.csv để tìm mọi resource_id, system_id, và lịch sử thao tác 
        của tập hợp các email (hiện tại + lịch sử) trước mốc audit_at.
        """

                    'details': row['details']
                })

        """
        Phân tích chuỗi activity của một user để xác định resource nào còn tồn tại
        hay đã bị xóa/ẩn danh trước audit_at.
        """

        # Sắp xếp theo thời gian tăng dần
        sorted_acts = sorted(activities, key=lambda x: x['event_time'])
        
        for act in sorted_acts:
            res = act['resource_id']
            ev_type = act['event_type']
            sys_target = act['target_system'] or act['source_system']
            
            if res not in resources_state:
                resources_state[res] = {
                    'system_id': sys_target,
                    'status': 'PRESENT',
                    'last_event': ev_type,
                    'evidence_id': act['activity_id']
                }

                    )
            elif ev_type in ['INDEX_RECORD', 'SYNC_RECORD', 'UPDATE_RECORD']:
                resources_state[res]['status'] = 'PRESENT'
                
        return resources_state

class UnstructuredInspector:
    def __init__(self, dataset_dir="dataset"):
        self.evidence_dir = os.path.join(dataset_dir, "evidence")
        
    def extract_text_from_pdf(self, pdf_path):
        """Trích xuất văn bản từ file PDF (hỗ trợ pypdf hoặc đọc thô)"""
        text_content = ""
        try:
            # Thử dùng pypdf nếu có sẵn
            import pypdf
            reader = pypdf.PdfReader(pdf_path)
            for page in reader.pages:
                t = page.extract_text()
                if t:
                    text_content += t + "\n"
        except Exception:
            # Fallback: Đọc nhị phân tìm chuỗi văn bản thô nếu thiếu thư viện
            try:
                with open(pdf_path, 'rb') as f:
                    content = f.read()
                    # Lọc các đoạn text ASCII/UTF-8 cơ bản trong PDF
                    text_content = ''.join([chr(b) if 32 <= b <= 126 else ' ' for b in content])
            except:
                pass
        return text_content.lower()

    def scan_unstructured_evidence(self, identifier):
        """
        Quét toàn bộ file PDF, TXT, LOG trong thư mục evidence 
        xem có chứa identifier (email, user_id) hay không.
        """
        if not os.path.exists(self.evidence_dir):
            return []
            
        matches = []
        clean_id = str(identifier).strip().lower()
        if not clean_id:
            return matches
            
        # 1. Quét các file PDF và text/log
        for ext in ["*.pdf", "*.txt", "*.log", "*.csv"]:
            for filepath in glob.glob(os.path.join(self.evidence_dir, "**", ext), recursive=True):
                filename = os.path.basename(filepath)
                content = ""
                
                if ext == "*.pdf":
                    content = self.extract_text_from_pdf(filepath)
                else:
                    try:
                        with open(filepath, 'r', encoding='utf-8', errors='ignore') as f:
                            content = f.read().lower()
                    except:
                        continue
                        
                if clean_id in content:
                    matches.append({
                        'file_name': filename,
                        'file_type': ext.replace("*.", ""),
                        'path': filepath
                    })
                    
        return matches

class SQLiteDeepInspector:
    def __init__(self, dataset_dir="dataset"):
        self.databases_dir = os.path.join(dataset_dir, "evidence", "databases")
        
    def compute_email_hashes(self, email):
        if not email or pd.isna(email):
            return []
        clean_email = str(email).strip().lower()
        return [
            hashlib.sha256(clean_email.encode()).hexdigest(),
            hashlib.md5(clean_email.encode()).hexdigest()
        ]

    def deep_query_snapshot(self, sql_path, user_id, email):
        conn = sqlite3.connect(":memory:")
        conn.row_factory = sqlite3.Row
        cursor = conn.cursor()
        found_hits = []
        try:
            with open(sql_path, 'r', encoding='utf-8', errors='ignore') as f:
                cursor.executescript(f.read())
                
            cursor.execute("SELECT name FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%';")
            tables = [row['name'] for row in cursor.fetchall()]
            hashes = self.compute_email_hashes(email)
            
            for table in tables:
                cursor.execute(f"PRAGMA table_info({table});")
                columns = [col['name'] for col in cursor.fetchall()]
                conditions, params = [], []
                
                if user_id and pd.notna(user_id):
                    for col in columns:
                        conditions.append(f"CAST({col} AS TEXT) = ?")
                        params.append(str(user_id))
                        
                if email and pd.notna(email):
                    for col in columns:
                        if any(k in col.lower() for k in ['email', 'text', 'profile', 'data', 'name']):
                            conditions.append(f"CAST({col} AS TEXT) LIKE ?")
                            params.append(f"%{email}%")
                            
                for h in hashes:
                    for col in columns:
                        if any(k in col.lower() for k in ['hash', 'id', 'key']):
                            conditions.append(f"CAST({col} AS TEXT) = ?")
                            params.append(h)
                            
                if not conditions:
                    continue
                    
                query = f"SELECT * FROM {table} WHERE " + " OR ".join(conditions)
                cursor.execute(query, params)
                for row in cursor.fetchall():
                    found_hits.append({'table': table, 'record': dict(row)})
        except Exception:
            pass
        finally:
            conn.close()
        return found_hits

    def search_all_databases(self, user_id, email):
        sql_files = glob.glob(os.path.join(self.databases_dir, "*.sql"))
        all_results = {}
        for sql_path in sql_files:
            filename = os.path.basename(sql_path)
            hits = self.deep_query_snapshot(sql_path, user_id, email)
            if hits:
                all_results[filename] = hits
        return all_results
